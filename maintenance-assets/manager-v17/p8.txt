function buildDailyReport(x){
  const {activity,open,emergencies,waiting,todayNew,completedToday}=x;
  const date=fmtDate(new Date(),false);
  const lines=[`*تقرير سير أعمال الصيانة – ${date}*`,'',`الطلبات الجديدة اليوم: ${todayNew.length}`,`تم التنفيذ اليوم: ${completedToday.length}`,`الطلبات المفتوحة: ${open.length}`,`الطلبات الطارئة المفتوحة: ${emergencies.length}`,`بانتظار قطعة / جهة خارجية: ${waiting.length}`];
  if(completedToday.length){lines.push('','*أعمال تم تنفيذها اليوم:*');completedToday.slice(0,12).forEach((r,i)=>lines.push(`${i+1}. رقم الطلب: ${ticketNumber(r.ticket_code)} | ${r.department} | ${r.description} ✅`));}
  const meaningful=activity.filter(a=>a.to_status && a.to_status!=='completed');
  if(meaningful.length){lines.push('','*تحديثات اليوم:*');meaningful.slice(0,15).forEach((a,i)=>lines.push(`${i+1}. رقم الطلب: ${ticketNumber(a.ticket_code)} | ${a.department} | ${STATUS[a.to_status]||a.to_status} — ${a.note}`));}
  if(emergencies.length){lines.push('','*طلبات طارئة ما زالت مفتوحة:*');emergencies.slice(0,10).forEach((r,i)=>lines.push(`${i+1}. رقم الطلب: ${ticketNumber(r.ticket_code)} | ${r.department} | ${r.description}`));}
  lines.push('','— نظام متابعة الصيانة');
  return lines.join('\n');
}
