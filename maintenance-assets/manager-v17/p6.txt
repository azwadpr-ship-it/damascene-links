async function loadAttachments(requestId) {
  const at=await api(`/rest/v1/request_attachments?request_id=eq.${requestId}&select=*&order=created_at.asc`);
  const out=[];
  for(const a of at||[]){
    try { const blob=await apiBlob(`/storage/v1/object/authenticated/maintenance-images/${a.storage_path}`); out.push({...a,url:URL.createObjectURL(blob)}); } catch { out.push({...a,url:null}); }
  }
  return out;
}
