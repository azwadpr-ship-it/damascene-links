async function submitRequest(e) {
  e.preventDefault(); const btn=e.submitter; btn.disabled=true; btn.textContent='جاري حفظ الطلب...';
  try {
    const payload={
      department:document.getElementById('department').value,
      location:document.getElementById('locationField').value.trim()||null,
      category:document.getElementById('category').value,
      description:document.getElementById('description').value.trim(),
      priority:document.getElementById('priority').value,
      requested_by:state.user.id
    };
    const rows=await api('/rest/v1/maintenance_requests?select=*',{method:'POST',headers:{Prefer:'return=representation'},body:JSON.stringify(payload)});
    const req=rows[0];
    try{await notifyMaintenanceNewRequest(req.id);}catch(e){console.warn(e);}
    const files=[...document.getElementById('requestFiles').files];
    let failedUploads=0;
    for(const file of files){try{await uploadAttachment(req,file,'request',null);}catch{failedUploads++;}}
    showShareModal(req,failedUploads);
    if(failedUploads) toast(`تم حفظ الطلب، لكن تعذر رفع ${failedUploads} مرفق. لا تعِد إرسال الطلب.`);
  } catch(ex) { toast(`تعذر إنشاء الطلب: ${ex.message}`); }
  finally { btn.disabled=false; btn.textContent='حفظ الطلب والمتابعة للمشاركة'; }
}
async function uploadAttachment(req,file,kind='request',updateId=null) {
  const safe=(file.name||'image.jpg').replace(/[^a-zA-Z0-9._-]+/g,'_');
  const path=`${req.id}/${Date.now()}_${Math.random().toString(36).slice(2,8)}_${safe}`;
  await api(`/storage/v1/object/maintenance-images/${path}`,{method:'POST',headers:{'Content-Type':file.type||'application/octet-stream','x-upsert':'false'},body:file});
  await api('/rest/v1/request_attachments',{method:'POST',headers:{Prefer:'return=minimal'},body:JSON.stringify({request_id:req.id,update_id:updateId,uploaded_by:state.user.id,kind,storage_path:path,file_name:file.name||null,mime_type:file.type||null})});
}
function shareText(req) {
  const p=PRIORITY[req.priority]||{label:req.priority,icon:''};
  return `${p.icon} *طلب صيانة ${req.priority==='emergency'?'طارئ':req.priority==='urgent'?'عاجل':'جديد'}*\n\nرقم الطلب: ${ticketNumber(req.ticket_code)}\nالقسم: ${req.department}\n${req.location?`الموقع: ${req.location}\n`:''}المشكلة: ${req.description}\nالأولوية: ${p.label}\nوقت البلاغ: ${fmtDate(req.created_at)}\nمقدم الطلب: ${state.profile.full_name}\n\nمتابعة الطلب:\nhttps://azwad-m.vercel.app/${ticketNumber(req.ticket_code)}`;
}
function showShareModal(req,failedUploads=0) {
  const d=document.createElement('div'); d.className='modal-backdrop'; d.id='shareModal';
  d.innerHTML=`<div class="modal"><h3>تم تسجيل الطلب رقم ${esc(ticketNumber(req.ticket_code))} ✅</h3><p>الخطوة الأخيرة: شارك الطلب في مجموعة الصيانة على واتساب حتى يصل التنبيه فورياً.</p>${failedUploads?`<div class="alert error">تم حفظ الطلب بنجاح، لكن تعذر رفع ${failedUploads} مرفق. لا تعِد إرسال الطلب.</div>`:''}${priorityBadge(req.priority)}<div class="actions"><a id="waShare" class="btn whatsapp block" target="_blank" rel="noopener">مشاركة الطلب على واتساب</a><button id="afterShare" class="btn ghost block" disabled>تمت المشاركة — العودة للنموذج</button></div></div>`;
  document.body.appendChild(d);
  const wa=document.getElementById('waShare'); wa.href=`https://wa.me/?text=${encodeURIComponent(shareText(req))}`;
  wa.addEventListener('click',()=>{setTimeout(()=>{const b=document.getElementById('afterShare');if(b)b.disabled=false;},900);});
  document.getElementById('afterShare').addEventListener('click',()=>{d.remove();renderRequestPage();});
}
