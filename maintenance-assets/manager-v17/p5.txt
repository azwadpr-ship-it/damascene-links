async function uploadAttachment(req,file,kind='request',updateId=null) {
  const safe=(file.name||'image.jpg').replace(/[^a-zA-Z0-9._-]+/g,'_');
  const path=`${req.id}/${Date.now()}_${Math.random().toString(36).slice(2,8)}_${safe}`;
  await api(`/storage/v1/object/maintenance-images/${path}`,{method:'POST',headers:{'Content-Type':file.type||'application/octet-stream','x-upsert':'false'},body:file});
  await api('/rest/v1/request_attachments',{method:'POST',headers:{Prefer:'return=minimal'},body:JSON.stringify({request_id:req.id,update_id:updateId,uploaded_by:state.user.id,kind,storage_path:path,file_name:file.name||null,mime_type:file.type||null})});
}

function sortMaintenanceRows(rows=[]){
  const rank={emergency:0,urgent:1,normal:2,improvement:3};
  return [...rows].sort((a,b)=>{
    const ac=['completed','cancelled'].includes(a.status)?1:0, bc=['completed','cancelled'].includes(b.status)?1:0;
    if(ac!==bc) return ac-bc;
    const p=(rank[a.priority]??9)-(rank[b.priority]??9);
    return p || new Date(a.created_at)-new Date(b.created_at);
  });
}
function ticketList(rows=[]) {
  if(!rows?.length) return '<div class="empty">لا توجد طلبات حتى الآن.</div>';
  return `<div class="ticket-list">${rows.map(r=>{
    const cls=r.priority==='emergency'?' ticket-emergency':r.priority==='urgent'?' ticket-urgent':'';
    const att=r.priority==='emergency'?'<span class="attention-label emergency-attention">اهتمام فوري</span>':r.priority==='urgent'?'<span class="attention-label urgent-attention">عاجل</span>':'';
    const name=r.requester_name || (r.requested_by===state.user?.id ? state.profile?.full_name : '—');
    return `<a href="/ticket/${r.id}" data-nav class="ticket${cls}">
      <div class="ticket-main">
        <h4 class="ticket-requester">رقم الطلب: ${esc(ticketNumber(r.ticket_code||''))} — مقدم الطلب: ${esc(name||'—')}</h4>
        <p class="ticket-department"><strong>القسم:</strong> ${esc(r.department)}${r.location?` — <strong>الموقع:</strong> ${esc(r.location)}`:''}</p>
        <p class="ticket-problem"><strong>المشكلة:</strong> ${esc(r.description)}</p>
        <p class="ticket-date">${fmtDate(r.created_at)}</p>
      </div>
      <div class="ticket-meta">${att}${priorityBadge(r.priority)} ${statusBadge(r.status)}</div>
    </a>`;
  }).join('')}</div>`;
}
function maintenanceListHtml(rows=[]){
  if(!rows?.length) return '<div class="empty">لا توجد طلبات حتى الآن.</div>';
  const hot=rows.filter(r=>!['completed','cancelled'].includes(r.status)&&['emergency','urgent'].includes(r.priority));
  if(!hot.length) return ticketList(rows);
  const ids=new Set(hot.map(r=>r.id)), rest=rows.filter(r=>!ids.has(r.id));
  return `<div class="priority-section"><div class="priority-section-title">⚠️ طلبات تحتاج اهتمام مباشر</div>${ticketList(hot)}</div>${rest.length?`<div><div class="regular-section-title">باقي الطلبات</div>${ticketList(rest)}</div>`:''}`;
}
